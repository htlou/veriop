  |- forall k n w a l1 l2 h1 h2 c1 c2.
         (n * w + 1 == 0) (mod (2 EXP 64)) /\
         l1 < 2 EXP (64 * k) /\
         l2 < 2 EXP (64 * k) /\
         h1 < 2 EXP (64 * k) /\
         h2 < 2 EXP (64 * k) /\
         n * l1 + a = 2 EXP (64 * k) * (2 EXP (64 * k) * c1 + h1) /\
         n * l2 + a = 2 EXP (64 * k) * (2 EXP (64 * k) * c2 + h2)
         ==> l1 = l2 /\ h1 = h2 /\ c1 = c2
